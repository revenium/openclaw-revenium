```bash
get_offset() {
  local sid="$1"
  if [[ ! -f "${OFFSETS_FILE}" ]]; then
    echo 0
    return
  fi
  python3 -c "
import json, sys
try:
    d = json.load(open('${OFFSETS_FILE}'))

except:
    print(0)
" 2>/dev/null || echo 0
}
```

2. `set_offset()` - takes session_id and line_count, uses python3 to atomically update the offsets file (read existing, update key, write to temp file, mv to final path):
```bash
set_offset() {
  local sid="$1"
  local count="$2"
  python3 -c "
import json, os, tempfile
path = '${OFFSETS_FILE}'
try:
    d = json.load(open(path))
except:
    d = {}
d['${sid}'] = int('${count}')
fd, tmp = tempfile.mkstemp(dir=os.path.dirname(path) or '.')
with os.fdopen(fd, 'w') as f:
    json.dump(d, f)
os.rename(tmp, path)
" 2>/dev/null || true
}
```
